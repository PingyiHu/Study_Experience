# **绝对位置编码**

```python
class PositionalEncoder(nn.Module):
    def __init__(self, d_model, max_seq_len = 80):
        super().__init__()
        self.d_model = d_model
        # 根据 pos 和 i 创建一个常量 PE 矩阵
        pe = torch.zeros(max_seq_len, d_model)
        for pos in range(max_seq_len):
            for i in range(0, d_model, 2):
                pe[pos, i] = math.sin(pos / (10000 ** ((2 * i)/d_model)))
                pe[pos, i + 1] = math.cos(pos / (10000 ** ((2 * (i + 1))/d_model)))
        pe = pe.unsqueeze(0)
        self.register_buffer('pe', pe)
    def forward(self, x):
        # 使得单词嵌入表示相对大一些
        x = x * math.sqrt(self.d_model)
        # 增加位置常量到单词嵌入表示中
        seq_len = x.size(1)
        x = x + Variable(self.pe[:,:seq_len], requires_grad=False).cuda()
        return x
```

# 写一个X^3=18的随机梯度下降解法

```python
import numpy as np

# 定义误差函数
def loss(x):
    return (x**3 - 18)**2

# 定义梯度
def grad(x):
    return 6 * x**2 * (x**3 - 18)

# 梯度下降参数
learning_rate = 0.001
max_iters = 10000
tolerance = 1e-6

# 初始化 x
x = np.random.randn()  # 随机起点
print(f"初始 x = {x:.4f}")

# 梯度下降主循环
for i in range(max_iters):
    dx = grad(x)
    x_new = x - learning_rate * dx
    # 如果对误差没有要求就用下面的情况结束迭代
    if abs(x_new - x) < tolerance:
        print(f"收敛于第 {i} 次迭代")
        break
    #如果对误差有要求就计算和目标的距离
    #if abs(x_next**3 - a) <= target:
    #    print(f"收敛于第 {i} 次迭代")
    #    break
    x = x_new

print(f"最终解：x = {x:.6f}")
print(f"验证：x^3 = {x**3:.6f}")
```

# 写一个X^3=18的牛顿法

```python
import numpy as np
def cbrt_newton(a, tol=1e-12, max_iter=100):
    # 处理零的特例
    if a == 0:
        return 0.0
    # 安全初值（避免 0，量级与 a 匹配）
	  x = np.random.randn()  # 随机起点
    x = x if a > 0 else -x

    target = tol * max(1.0, abs(a))
    for _ in range(max_iter):
        # 牛顿步：x_{k+1} = (2x + a/x^2)/3
        x_next = (2.0 * x + a / (x * x)) / 3.0
        if abs(x_next**3 - a) <= target:
            return x_next
        x = x_next
    return x  # 达到迭代上限时返回当前最优
```

```python
def cross_entropy_loss(y_true, y_pred):
    y_pred = np.clip(y_pred, 1e-9, 1 - 1e-9)
    return -np.sum(y_true * np.log(y_pred)) / y_true.shape[0]

# Example usage
y_true = np.array([[1, 0], [0, 1]])  # One-hot encoding
y_pred = np.array([[0.8, 0.2], [0.1, 0.9]])  # Model output (probabilities)
print(cross_entropy_loss(y_true, y_pred))  # Expected: Small loss value
```

```python
import numpy as np

def kl_divergence_probs(p, q, axis=1):
    """
    计算 KL(p || q)，p,q 为批量的类别分布（每行和为1）
    返回 batch 平均 KL
    """
    eps = 1e-9

    # 避免 log(0)
    p = np.clip(p, eps, 1.0)
    q = np.clip(q, eps, 1.0)

    per_sample = (p * (np.log(p) - np.log(q))).sum(axis=axis)
    return per_sample.mean()
```

# **K-Means 聚类**

```python
class KMeans:
    def __init__(self, k=2, max_iters=100):
        self.k = k
        self.max_iters = max_iters

    def fit(self, X):
        n_samples, n_features = X.shape
        self.centroids = X[np.random.choice(n_samples, self.k, replace=False)]

        for _ in range(self.max_iters):
            clusters = [np.argmin([np.linalg.norm(x - c) for c in self.centroids]) for x in X]
            new_centroids = np.array([X[np.array(clusters) == i].mean(axis=0) for i in range(self.k)])

            if np.all(new_centroids == self.centroids):
                break
            self.centroids = new_centroids

    def predict(self, X):
        return np.array([np.argmin([np.linalg.norm(x - c) for c in self.centroids]) for x in X])

# Example usage
X = np.array([[1, 2], [1, 4], [5, 6], [8, 8], [1, 0], [6, 5]])
kmeans = KMeans(k=2)
kmeans.fit(X)
print(kmeans.predict(X))  # Expected: Cluster labels (0 or 1)
```

# **线性回归**

```python
import numpy as np

class LinearRegression:
    def __init__(self, lr=0.01, epochs=1000):
        self.lr = lr
        self.epochs = epochs
        self.W = None
        self.b = None

    def fit(self, X, y):
        n_samples, n_features = X.shape
        self.W = np.zeros(n_features)
        self.b = 0

        for _ in range(self.epochs):
            y_pred = np.dot(X, self.W) + self.b
            dW = (1 / n_samples) * np.dot(X.T, (y_pred - y))
            db = (1 / n_samples) * np.sum(y_pred - y)

            self.W -= self.lr * dW
            self.b -= self.lr * db

    def predict(self, X):
        return np.dot(X, self.W) + self.b

# Example usage
X = np.array([[1], [2], [3], [4]])
y = np.array([2, 4, 6, 8])
lr = LinearRegression()
lr.fit(X, y)
print(lr.predict(np.array([[5]])))  # Expected: ~10
```
