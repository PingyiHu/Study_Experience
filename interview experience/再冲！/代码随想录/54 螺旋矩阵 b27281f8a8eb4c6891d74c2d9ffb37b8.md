# 54. 螺旋矩阵

Tags: 模拟
URL: https://leetcode.cn/problems/spiral-matrix/description/

```cpp
class Solution
{
public:
    vector<int> spiralOrder(vector<vector<int>> &matrix)
    {
        int width = matrix.size();
        int length = matrix[0].size();
        int total = width * length;
        vector<int> result;
        int count = 0;
        int u = 0, d = width - 1, l = 0, r = length - 1;
        while (count < total)
        {
            for (int i = l; i <= r; i++)
            {
                count++;
                result.push_back(matrix[u][i]);
            }
            u++;
            for (int i = u; i <= d; i++)
            {
                count++;
                result.push_back(matrix[i][r]);
            }
            r--;
            for (int i = r; i >= l; i--)
            {
                count++;
                result.push_back(matrix[d][i]);
            }
            d--;
            for (int i = d; i >= u; i--)
            {
                count++;
                result.push_back(matrix[i][l]);
            }
            l++;
        }
        result.erase(result.begin() + total, result.end());
        return result;
    }
};
```
