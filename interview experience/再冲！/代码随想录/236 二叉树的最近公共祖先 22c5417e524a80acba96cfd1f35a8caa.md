# 236. 二叉树的最近公共祖先

Tags: 二叉树
URL: https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/description/

# 我的方法

后序遍历，拿一个pair装当前p或者q是否在某一个子树上

```cpp
class Solution
{
public:
    TreeNode *parent = NULL;
    void postorder(TreeNode *root, pair<bool, bool> &flag, TreeNode *p, TreeNode *q)
    {
        if (!root)
            return;

        pair<bool, bool> l_flag = make_pair(false, false);
        postorder(root->left, l_flag, p, q);

        pair<bool, bool> r_flag = make_pair(false, false);
        postorder(root->right, r_flag, p, q);

        if (p == root)
            flag.first = true;

        if (q == root)
            flag.second = true;

        bool l = l_flag.first || r_flag.first || flag.first;
        bool r = l_flag.second || r_flag.second || flag.second;

        if (l && r)
            parent = root;
        else
        {
            flag.first = l;
            flag.second = r;
        }
    }

    TreeNode *lowestCommonAncestor(TreeNode *root, TreeNode *p, TreeNode *q)
    {
        pair<bool, bool> init_flag = make_pair(false, false);
        postorder(root, init_flag, p, q);
        return parent;
    }
};
```

# 随想录

后序的方式进行回溯

- 如果p子树上面有q的话，这个时候只需要返回p即可
- 如果p和q在不同的树上，这时返回公共节点

```cpp
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root == q || root == p || root == NULL) return root;
        TreeNode* left = lowestCommonAncestor(root->left, p, q);
        TreeNode* right = lowestCommonAncestor(root->right, p, q);
        if (left != NULL && right != NULL) return root;

        if (left == NULL && right != NULL) return right;
        else if (left != NULL && right == NULL) return left;
        else  { //  (left == NULL && right == NULL)
            return NULL;
        }

    }
};
```
