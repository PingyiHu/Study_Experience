```python
class Solution(object):
    def numIslands(self, grid):
        """
        :type grid: List[List[str]]
        :rtype: int
        """
        result = 0
        visited = [[False for _ in range(len(grid[0]))] for _ in range(len(grid))]

        def dfs(grid, visited, i, j):
            if visited[i][j] == True:
                return

            visited[i][j] = True
            if i - 1 >= 0 and grid[i - 1][j] == "1":
                dfs(grid, visited, i - 1, j)

            if j - 1 >= 0 and grid[i][j - 1] == "1":
                dfs(grid, visited, i, j - 1)

            if i + 1 < len(grid) and grid[i + 1][j] == "1":
                dfs(grid, visited, i + 1, j)

            if j + 1 < len(grid[0]) and grid[i][j + 1] == "1":
                dfs(grid, visited, i, j + 1)

        for i in range(len(grid)):
            for j in range(len(grid[i])):
                if visited[i][j] != True and grid[i][j] == "1":
                    dfs(grid, visited, i, j)
                    result += 1
        return result
```

## bfs

使用栈和队列都可以，只不过可能第一圈顺时针扩张第二圈是逆时针扩张

```python
class Solution(object):
    def numIslands(self, grid):
        """
        :type grid: List[List[str]]
        :rtype: int
        """
        result = 0
        visited = [[False for _ in range(len(grid[0]))] for _ in range(len(grid))]
        dirs = [(-1, 0), (0, 1), (1, 0), (0, -1)]

        def bfs(grid, visited, i, j):
            stack = [(i, j)]
            while stack:
                x = stack[-1][0]
                y = stack[-1][1]
                stack.pop()
                for dir in dirs:
                    cur_x = x + dir[0]
                    cur_y = y + dir[1]
                    if (
                        0 <= cur_x < len(grid)
                        and 0 <= cur_y < len(grid[0])
                        and grid[cur_x][cur_y] == "1"
                        and visited[cur_x][cur_y] == False
                    ):
                        visited[cur_x][cur_y] = True
                        stack.append((cur_x, cur_y))

        for i in range(len(grid)):
            for j in range(len(grid[i])):
                if visited[i][j] == False and grid[i][j] == "1":
                    visited[i][j] = True
                    bfs(grid, visited, i, j)
                    result += 1
        return result
```

```python
direction = [[0, 1], [1, 0], [0, -1], [-1, 0]]  # 四个方向：上、右、下、左

def dfs(grid, visited, x, y):
    """
    对一块陆地进行深度优先遍历并标记
    """
    # 与版本一的差别，在调用前增加判断终止条件
    if visited[x][y] or grid[x][y] == 0:
        return
    visited[x][y] = True

    for i, j in direction:
        next_x = x + i
        next_y = y + j
        # 下标越界，跳过
        if next_x < 0 or next_x >= len(grid) or next_y < 0 or next_y >= len(grid[0]):
            continue
        # 由于判断条件放在了方法首部，此处直接调用dfs方法
        dfs(grid, visited, next_x, next_y)

if __name__ == '__main__':
    # 版本二
    n, m = map(int, input().split())

    # 邻接矩阵
    grid = []
    for i in range(n):
        grid.append(list(map(int, input().split())))

    # 访问表
    visited = [[False] * m for _ in range(n)]

    res = 0
    for i in range(n):
        for j in range(m):
            # 判断：如果当前节点是陆地，res+1并标记访问该节点，使用深度搜索标记相邻陆地。
            if grid[i][j] == 1 and not visited[i][j]:
                res += 1
                dfs(grid, visited, i, j)

    print(res)
```

```python
from collections import deque
directions = [[0, 1], [1, 0], [0, -1], [-1, 0]]
def bfs(grid, visited, x, y):
    que = deque([])
    que.append([x,y])
    visited[x][y] = True
    while que:
        cur_x, cur_y = que.popleft()
        for i, j in directions:
            next_x = cur_x + i
            next_y = cur_y + j
            if next_y < 0 or next_x < 0 or next_x >= len(grid) or next_y >= len(grid[0]):
                continue
            if not visited[next_x][next_y] and grid[next_x][next_y] == 1: 
                visited[next_x][next_y] = True
                que.append([next_x, next_y])

def main():

        grid.append(list(map(int, input().split())))

    res = 0
    for i in range(n):
        for j in range(m):
            if grid[i][j] == 1 and not visited[i][j]:
                res += 1
                bfs(grid, visited, i, j)
    print(res)

if __name__ == "__main__":
    main()
```
