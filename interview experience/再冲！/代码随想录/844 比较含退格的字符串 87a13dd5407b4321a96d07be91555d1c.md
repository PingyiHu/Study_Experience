# 844. 比较含退格的字符串

Tags: 双指针
URL: https://leetcode.cn/problems/backspace-string-compare/description/

牢记快慢指针的定义，他们分别代表什么。我的方法如下，不如用一个栈。

```cpp
class Solution
{
public:
    bool backspaceCompare(std::string s, std::string t)
    {
        int len_s = afterBackspace(s);
        int len_t = afterBackspace(t);
        if (len_s != len_t)
            return false;
        int i = 0;
        while (i < len_s)
        {
            if (s[i] != t[i])
                return false;
            i++;
        }
        return true;
    }

private:
    int afterBackspace(std::string &s)
    {
        int i = 0; // slow,指向当前的新数组元素
        int j = 0; // fast
        for (; j < s.size(); j++)
        {
            if (s[j] != '#')
            {
                s[i]=s[j];
                i++;
                continue;
            }
            while (j < s.size() && s[j] == '#')
            {
                j++;
                i--;
            }
            if (j < s.size())
            {
                if (i < 0)
                    i = 0;
                s[i] = s[j];
                i++;
            }
        }
        return i;
    }
};
```

官方题解，双指针法，由后往前找对应。

```cpp
class Solution {
public:
    bool backspaceCompare(string S, string T) {
        int i = S.length() - 1, j = T.length() - 1;
        int skipS = 0, skipT = 0;

        while (i >= 0 || j >= 0) {
            while (i >= 0) {
                if (S[i] == '#') {
                    skipS++, i--;
                } else if (skipS > 0) {
                    skipS--, i--;
                } else {
                    break;
                }
            }
            while (j >= 0) {
                if (T[j] == '#') {
                    skipT++, j--;
                } else if (skipT > 0) {
                    skipT--, j--;
                } else {
                    break;
                }
            }
            if (i >= 0 && j >= 0) {
                if (S[i] != T[j]) {
                    return false;
                }
            } else {
                if (i >= 0 || j >= 0) {
                    return false;
                }
            }
            i--, j--;
        }
        return true;
    }
};
```
