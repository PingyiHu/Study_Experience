# 541. 反转字符串II

Tags: 字符串
URL: https://leetcode.cn/problems/reverse-string-ii/

# 我的：双指针

```cpp
class Solution {
public:
    string reverseStr(string s, int k) {
        int round = s.size() / (2 * k);
        int remain = s.size() % (2 * k);
        for (int i = 0; i < round; i++) {
            int j = i * (2 * k);
            int l = j + k - 1;
            while (j < l) {
                swap(s[j], s[l]);
                j++;
                l--;
            }
        }
        if (remain < k) {
            int j = round * 2 * k;
            int l = s.size() - 1;
            while (j < l) {
                swap(s[j], s[l]);
                j++;
                l--;
            }
        } else {
            int j = round * 2 * k;
            int l = j + k - 1;
            while (j < l) {
                swap(s[j], s[l]);
                j++;
                l--;
            }
        }
        return s;
    }
};
```

# 随想录

每次移动2k个元素

```cpp
class Solution {
public:
    void reverse(string& s, int start, int end) {
        for (int i = start, j = end; i < j; i++, j--) {
            swap(s[i], s[j]);
        }
    }
    string reverseStr(string s, int k) {
        for (int i = 0; i < s.size(); i += (2 * k)) {
            // 1. 每隔 2k 个字符的前 k 个字符进行反转
            // 2. 剩余字符小于 2k 但大于或等于 k 个，则反转前 k 个字符
            if (i + k <= s.size()) {
                reverse(s, i, i + k - 1);
                continue;
            }
            // 3. 剩余字符少于 k 个，则将剩余字符全部反转。
            reverse(s, i, s.size() - 1);
        }
        return s;
    }
};
```
