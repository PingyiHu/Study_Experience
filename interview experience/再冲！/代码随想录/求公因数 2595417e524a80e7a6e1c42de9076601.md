# 求公因数

```python
import math

def divisors(n: int) -> list[int]:
    if n <= 0:
        raise ValueError("n 必须是正整数")
    res = set()
    r = int(math.isqrt(n))  # 避免浮点误差
    for i in range(1, r + 1):
        if n % i == 0:
            res.add(i)
            res.add(n // i)  # 成对加入
    return sorted(res)
```
