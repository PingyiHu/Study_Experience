# 112. 路径总和

Tags: 二叉树
URL: https://leetcode.cn/problems/path-sum/description/

# 我的方法

使用中序遍历，到叶子节点后进行比较。

在使用非递归的方式解题时，需要将递归时传入的参数，进行压栈，保证参数进栈和退栈的顺序和节点的顺序相同。

```cpp
class Solution {
public:
    bool hasPathSum(TreeNode* root, int targetSum) {
        if (root == NULL)
            return false;
        stack<pair<TreeNode*, bool>> st;
        stack<int> sum_st;
        st.push(make_pair(root, false));
        sum_st.push(root->val);
        while (!st.empty()) {
            auto node = st.top().first;
            auto flag = st.top().second;
            int now_sum = sum_st.top();
            st.pop();
            sum_st.pop();
            if (flag) {
                if (now_sum == targetSum && node->left == NULL &&
                    node->right == NULL)
                    return true;
            } else {
                if (node->right) {
                    st.push(make_pair(node->right, false));
                    sum_st.push(now_sum + node->right->val);
                }

                if (node->left) {
                    st.push(make_pair(node->left, false));
                    sum_st.push(now_sum + node->left->val);
                }
                st.push(make_pair(node, true));
                sum_st.push(now_sum);
            }
        }
        return false;
    }
};
```
