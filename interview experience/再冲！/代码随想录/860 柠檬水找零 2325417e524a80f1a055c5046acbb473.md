# 860. 柠檬水找零

Tags: 贪心
URL: https://leetcode.cn/problems/lemonade-change/submissions/644577343/

# 我的方法

先把大钱破开，再给小钱，这样小钱多就可以还10块的情况

```cpp
class Solution {
public:
    bool lemonadeChange(vector<int>& bills) {
        int d_5 = 0;
        int d_10 = 0;
        for (int i = 0; i < bills.size(); i++) {
            if (bills[i] == 5)
                d_5++;
            else if (bills[i] == 10) {
                if (d_5 == 0)
                    return false;
                else {
                    d_5--;
                    d_10++;
                }
            } else {
                if (d_10 >= 1 && d_5 >= 1) {
                    d_5--;
                    d_10--;
                } else if (d_5 >= 3)
                    d_5 -= 3;
                else
                    return false;
            }
        }
        return true;
    }
};
```
