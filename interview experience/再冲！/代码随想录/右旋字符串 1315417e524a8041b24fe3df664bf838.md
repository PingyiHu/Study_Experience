# include<iostream>
# include<string>
using namespace std;

void reverse_single(string& s,int start,int end){
    while(start<end){
        swap(s[start++],s[end--]);
    }
}

int main(){
    int k=0;
    string s;
    cin>>k;
    cin>>s;
    reverse_single(s,0,s.size()-1);
    reverse_single(s,0,k-1);
    reverse_single(s,k,s.size()-1);
    cout<<s;
    return 0;
}
