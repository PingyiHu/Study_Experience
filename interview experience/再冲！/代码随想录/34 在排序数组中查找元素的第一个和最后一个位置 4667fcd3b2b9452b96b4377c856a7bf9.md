# 34. 在排序数组中查找元素的第一个和最后一个位置

Tags: 二分查找
URL: https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/description/

我的方法，先二分找一个target，再让left和right慢慢扩展。

```cpp
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int left = 0;
        int right = nums.size() - 1;
        int leftbound = -1;
        int rightbound = -1;
        int rflag = 1;
        int lflag = 1;
        int findflag=0;
        vector<int> a;
        if (right == -1) {
            a.push_back(leftbound);
            a.push_back(rightbound);
            return a;
        }
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] > target)
                right = mid - 1;
            else if (nums[mid] < target)
                left = mid + 1;
            else {
                left = mid;
                right = mid;
                findflag=1;
                break;
            }
        }
        if(!findflag){
            a.push_back(leftbound);
            a.push_back(rightbound);
            return a;
        }
        while (rflag || lflag) {
            if (left >= 0 && target == nums[left]) {
                leftbound = left;
                left--;
            } else {
                lflag = 0;
            }
            if (right <= nums.size() - 1 && target == nums[right]) {
                rightbound = right;
                right++;
            } else {
                rflag = 0;
            }
        }
        a.push_back(leftbound);
        a.push_back(rightbound);
        return a;
    }
};
```

随想录方法：使用二分法分别找左右边界。将中间值等于和小于target的情况合在一起：

- 相当于是找一个刚好大于target的数在数组中的插入位置。
- right始终位于target的右边。

```cpp
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int leftBorder = getLeftBorder(nums, target);
        int rightBorder = getRightBorder(nums, target);
        // 情况一
        if (leftBorder == -2 || rightBorder == -2) return {-1, -1};
        // 情况三
        if (rightBorder - leftBorder > 1) return {leftBorder + 1, rightBorder - 1};
        // 情况二
        return {-1, -1};
    }
private:
     int getRightBorder(vector<int>& nums, int target) {
        int left = 0;
        int right = nums.size() - 1;
        int rightBorder = -2; // 记录一下rightBorder没有被赋值的情况
        while (left <= right) {
            int middle = left + ((right - left) / 2);
            if (nums[middle] > target) {
                right = middle - 1;
            } else { // 寻找右边界，nums[middle] == target的时候更新left
                left = middle + 1;
                rightBorder = left;
            }
        }
        return rightBorder;
    }
    int getLeftBorder(vector<int>& nums, int target) {
        int left = 0;
        int right = nums.size() - 1;
        int leftBorder = -2; // 记录一下leftBorder没有被赋值的情况
        while (left <= right) {
            int middle = left + ((right - left) / 2);
            if (nums[middle] >= target) { // 寻找左边界，nums[middle] == target的时候更新right
                right = middle - 1;
                leftBorder = right;
            } else {
                left = middle + 1;
            }
        }
        return leftBorder;
    }
};
```
