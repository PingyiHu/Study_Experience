# dp数组的含义

背包一般是二维数组，只是有些情况下可以优化为1维的数组。

dp[i][j]：

- 代表着，[0, i]的物品任取放入容量为j的背包中，最大的价值
- i是物品
- j是背包容量

# 递推公式

第i个物品取或不取，比较最大的价值:

- 第i个物品可以取，要从不包含第i个物品，即[0, i)物品任取，背包容量为j-weight[i]：空出容量weight[i]，能取得的最大价值，和当前物品i价值相加。
- 第i个物品不取，那么就等于dp[i-1][j]，价值不变。

# 初始化

每个位置上的值都是由左上推导来的。因此初始化第一行和第一列就够了

- 背包大小为0的时候，什么都装不了
- 装第0个物品，背包的空间不断的增加，直到可以装下第一个物品，不然就置零

# 我的代码

时刻记住，j是空间，space[i]是空间；i是物品

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution
{
public:
    int bags(int N, vector<int> &space, vector<int> &value)
    {
        vector<vector<int>> dp(space.size(), vector<int>(N + 1, 0));
        for (int j = 1; j <= N; j++)
            if (space[0] <= j)
                dp[0][j] = value[0];
        for (int i = 1; i < space.size(); i++)
            for (int j = 1; j <= N; j++)
            {
                int no = dp[i - 1][j];
                int yes = 0;
                if (space[i] <= j)
                    yes = dp[i - 1][j - space[i]] + value[i];
                dp[i][j] = max(yes, no);
            }

        return dp[space.size() - 1][N];
    }
};

int main()
{
    int M, N;
    cin >> M >> N;
    vector<int> space(M);
    vector<int> value(M);
    for (int i = 0; i < M; i++)
        cin >> space[i];
    for (int i = 0; i < M; i++)
        cin >> value[i];
    Solution s;
    cout << s.bags(N, space, value);
    return 0;
}
```

# 随想录

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n, bagweight;// bagweight代表行李箱空间

    cin >> n >> bagweight;

    vector<int> weight(n, 0); // 存储每件物品所占空间
    vector<int> value(n, 0);  // 存储每件物品价值

    for(int i = 0; i < n; ++i) {
        cin >> weight[i];
    }
    for(int j = 0; j < n; ++j) {
        cin >> value[j];
    }
    // dp数组, dp[i][j]代表行李箱空间为j的情况下,从下标为[0, i]的物品里面任意取,能达到的最大价值
    vector<vector<int>> dp(weight.size(), vector<int>(bagweight + 1, 0));

    // 初始化, 因为需要用到dp[i - 1]的值
    // j < weight[0]已在上方被初始化为0
    // j >= weight[0]的值就初始化为value[0]
    for (int j = weight[0]; j <= bagweight; j++) {
        dp[0][j] = value[0];
    }

    for(int i = 1; i < weight.size(); i++) { // 遍历物品
        for(int j = 0; j <= bagweight; j++) { // 遍历行李箱容量
            if (j < weight[i]) 
		            dp[i][j] = dp[i - 1][j]; // 如果装不下这个物品,那么就继承dp[i - 1][j]的值
            else {
                dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - weight[i]] + value[i]);
            }
        }
    }
    cout << dp[n - 1][bagweight] << endl;

    return 0;
}
```

## 二维数组降为一维

原来的是按行进行遍历，用到的是上一行的内容。可以复用一个一维数组，注意按列从后往前进行遍历，这样就可以保证上一层：

- 在访问到i行j列的时候，i-1行j列的元素一定是没被覆盖的
- 在访问到i行j列的时候，i-1行j-weight[i]处一定也是没有被覆盖的

dp[i]是容量为i的背包装的最大价值。

```cpp
// 一维dp数组实现
#include <iostream>
#include <vector>
using namespace std;

int main() {
    // 读取 M 和 N
    int M, N;
    cin >> M >> N;

    vector<int> costs(M);
    vector<int> values(M);

    for (int i = 0; i < M; i++) {
        cin >> costs[i];
    }
    for (int j = 0; j < M; j++) {
        cin >> values[j];
    }

    // 创建一个动态规划数组dp，初始值为0
    vector<int> dp(N + 1, 0);

    // 外层循环遍历每个类型的研究材料
    for (int i = 0; i < M; ++i) {
        // 内层循环从 N 空间逐渐减少到**当前研究材料所占空间**
        for (int j = N; j >= costs[i]; --j) {
            // 考虑当前研究材料选择和不选择的情况，选择最大值
            dp[j] = max(dp[j], dp[j - costs[i]] + values[i]);
        }
    }

    // 输出dp[N]，即在给定 N 行李空间可以携带的研究材料最大价值
    cout << dp[N] << endl;

    return 0;
}
```
