# 501. 二叉搜索树中的众数

Tags: 二叉树
URL: https://leetcode.cn/problems/find-mode-in-binary-search-tree/

# 我的方法

二叉搜索树中序情况下，众数是连在一起的，所以，只要用一个计数器，记住当前最大的众数，以及其次数即可。

```cpp
class Solution
{
public:
    vector<int> result;
    int max_counter = 0;
    int counter = -1;
    int now_num = INT_MIN;
    void inorder(TreeNode *root)
    {
        if (root == NULL)
            return;
        inorder(root->left);

        if (root->val != now_num)
        {
            if (counter == max_counter)
            {
                result.push_back(now_num);
            }
            else if (counter > max_counter)
            {
                max_counter = counter;
                result.clear();
                result.push_back(now_num);
            }

            counter = 1;
            now_num = root->val;
        }
        else
            counter++;

        inorder(root->right);
    }

    vector<int> findMode(TreeNode *root)
    {
        inorder(root);
        if (counter == max_counter)
        {
            result.push_back(now_num);
        }
        else if (counter > max_counter)
        {
            max_counter = counter;
            result.clear();
            result.push_back(now_num);
        }
        return result;
    }
};
```

# 随想录

双指针法，弄一个pre

```cpp
class Solution {
private:
    int maxCount = 0; // 最大频率
    int count = 0; // 统计频率
    TreeNode* pre = NULL;
    vector<int> result;
    void searchBST(TreeNode* cur) {
        if (cur == NULL) return ;

        searchBST(cur->left);       // 左
                                    // 中
        if (pre == NULL) { // 第一个节点
            count = 1;
        } else if (pre->val == cur->val) { // 与前一个节点数值相同
            count++;
        } else { // 与前一个节点数值不同
            count = 1;
        }
        pre = cur; // 更新上一个节点

        if (count == maxCount) { // 如果和最大值相同，放进result中
            result.push_back(cur->val);
        }

        if (count > maxCount) { // 如果计数大于最大值频率
            maxCount = count;   // 更新最大频率
            result.clear();     // 很关键的一步，不要忘记清空result，之前result里的元素都失效了
            result.push_back(cur->val);
        }

        searchBST(cur->right);      // 右
        return ;
    }

public:
    vector<int> findMode(TreeNode* root) {
        count = 0;
        maxCount = 0;
        pre = NULL; // 记录前一个节点
        result.clear();

        searchBST(root);
        return result;
    }
};
```
