# 我的方法

求的是排列，需要先遍历容量，再遍历物品，保证物品可以多次被添加。

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution
{
public:
    int climb(int n, int m)
    {
        vector<int> dp(n + 1, 0);
        dp[0] = 1;
        for (int j = 1; j <= n; j++)
        { // 遍历背包容量
            for (int i = 1; i <= m && i <= j; i++)
            { // 遍历物品
                if (j - i >= 0)
                    dp[j] += dp[j - i];
            }
        }
        return dp[n];
    }
};

int main()
{
    int m, n;
    cin >> n >> m;
    Solution s;
    int ways = s.climb(n, m);
    cout << ways;
    return 0;
}
```
