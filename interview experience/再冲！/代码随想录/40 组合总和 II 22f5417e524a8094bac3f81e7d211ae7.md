# 40. 组合总和 II

Tags: 回溯

# 我的方法

要求的是：

- 组合不可以重复
- 元素可以重复
- 元素重复的次数有限

先进行排序，然后同层的相同元素不重复访问。

```cpp
class Solution
{
public:
    vector<vector<int>> result;
    vector<int> tmp;
    int sum = 0;

    void backtracking(vector<int> &candidates, int start, int sum, int target)
    {
        if (sum == target)
        {
            result.push_back(tmp);
            return;
        }

        for (int i = start; i < candidates.size() && candidates[i] + sum <= target; i++)
        {
            if (i > start && candidates[i] == candidates[i - 1])
                continue; //同层的相同元素不重复访问
            sum += candidates[i];
            tmp.push_back(candidates[i]);
            backtracking(candidates, i + 1, sum, target);
            sum -= candidates[i];
            tmp.pop_back();
        }
    }

    vector<vector<int>> combinationSum2(vector<int> &candidates, int target)
    {
        sort(candidates.begin(), candidates.end());
        backtracking(candidates, 0, 0, target);
        return result;
    }
};
```
