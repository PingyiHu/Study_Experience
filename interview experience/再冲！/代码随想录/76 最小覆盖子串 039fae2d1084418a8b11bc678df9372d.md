# 76. 最小覆盖子串

Tags: 双指针, 滑动窗口
URL: https://leetcode.cn/problems/minimum-window-substring/description/

力扣答案：

```cpp
class Solution {
public:
    unordered_map <char, int> ori, cnt;

    bool check() {
        for (const auto &p: ori) {
            if (cnt[p.first] < p.second) {
                return false;
            }
        }
        return true;
    }

    string minWindow(string s, string t) {
        for (const auto &c: t) {
            ++ori[c];
        }

        int l = 0, r = -1;
        int len = INT_MAX, ansL = -1, ansR = -1;

        while (r < int(s.size())) {
            if (ori.find(s[++r]) != ori.end()) {
                ++cnt[s[r]];
            }
            while (check() && l <= r) {
                if (r - l + 1 < len) {
                    len = r - l + 1;
                    ansL = l;
                }
                if (ori.find(s[l]) != ori.end()) {
                    --cnt[s[l]];
                }
                ++l;
            }
        }

        return ansL == -1 ? string() : s.substr(ansL, len);
    }
};
```

我的方法：讨论左右区间移动的时刻，和区间的内容。

```cpp
class Solution {
public:
    string minWindow(string s, string t) {
        // 子串比主串还要长直接返回
        if (s.size() < t.size())
            return "";

        // 初始化子串的字符要求
        unordered_map<char, int> request; // 要求的子串内容
        for (int i = 0; i < t.size(); i++) {
            request[t[i]]++;
        }
        int alter_flag = 0;
        // 滑动窗口
        int left = 0;  // 窗口的左边界
        int right = 0; // 窗口的右边界的后一位
        int result[2] = {0, INT32_MAX};
        unordered_map<char, int> nowChar; // 装入当前子串以及子串的个数。
        for (; right < s.size(); right++) {
            // 判断当前字符是不是在request中
            if (request.find(s[right]) != request.end()) {
                nowChar[s[right]]++;
                // 当前找到了所有的需要的字符，判断字符个数是否满足要求
                while (nowChar.size() == request.size()) {
                    int flag = 1;
                    for (auto it = request.begin(); it != request.end(); ++it) {
                        if (it->second > nowChar[it->first]) {
                            flag = 0;
                            break;
                        }
                    }
                    // 当前的序列满足字符个数要求,更新result,更新left。
                    if (flag) {
                        if (result[1] - result[0] > right - left) {
                            alter_flag = 1;
                            result[1] = right;
                            result[0] = left;
                        }
                        // left处是无关字符
                        if (request.find(s[left]) != request.end()) {
                            nowChar[s[left]]--;
                        }
                        left++;
                    }
                    // 字符不满足要求，继续向后查找
                    else
                        break;
                }
            }
        }
        if (alter_flag)
            return s.substr(result[0], result[1] - result[0] + 1);
        else
            return "";
    }
};
```

GPT4优化版本

1. **优化字符检查逻辑**：您当前的实现在缩小窗口时使用了一个内循环来检查**`nowChar`**是否满足**`request`**的要求。这个检查是通过遍历所有字符来实现的，当字符集较大时，这将导致较高的时间复杂度。
2. **减少不必要的哈希表查找**：您在每次右移窗口时都检查**`s[right]`**是否在**`request`**中，这是必要的。但是，您在左移窗口时也进行了相同的检查，实际上您可以仅在**`nowChar`**中更新计数时才做这个检查。
3. **使用一个变量跟踪满足条件的字符数**：您可以使用一个整数来跟踪当前窗口中满足**`t`**中字符数量要求的字符数。这样，您可以避免在每次窗口调整时重新计算满足条件的字符总数。

```cpp
class Solution {
public:
    string minWindow(string s, string t) {
        if (s.size() < t.size()) return "";

        unordered_map<char, int> charCountT;
        for (char c : t) charCountT[c]++;

        unordered_map<char, int> window;
        int have = 0;  // 窗口中满足要求数量的字符种类数
        int need = charCountT.size();  // 需要满足的字符种类总数
        int resLen = INT_MAX, resLeft = 0;

        int l = 0;
        for (int r = 0; r < s.size(); r++) {
            char c = s[r];
            window[c]++;
            // 只有当字符c是t中的字符，且数量恰好达到需求时，have才增加
            if (charCountT.count(c) && window[c] == charCountT[c])
                have++;

            // 当窗口满足所有字符需求时，尝试收缩左边界
            while (have == need) {
                // 更新结果
                if (r - l + 1 < resLen) {
                    resLen = r - l + 1;
                    resLeft = l;
                }
                // 减少左边界字符的计数
                window[s[l]]--;
                // 如果字符是必需的，并且窗口中的数量少于所需的数量，则have减少
                if (charCountT.count(s[l]) && window[s[l]] < charCountT[s[l]])
                    have--;
                l++;
            }
        }

        return resLen == INT_MAX ? "" : s.substr(resLeft, resLen);
    }
};
```
