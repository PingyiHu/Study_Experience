# 235. 二叉搜索树的最近公共祖先

Tags: 二叉树
URL: https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-search-tree/description/

# 我的方法

利用左小于根小于右的性质，如果当前的根是夹在两者数值之间，那么这个根就是公共祖先

```cpp
class Solution
{
public:
    TreeNode *postorder(TreeNode *root, TreeNode *p, TreeNode *q)
    {
        if (root == q || root == p)
            return root; //可以删除
            
        if (q->val < root->val)
            return postorder(root->left, p, q);
        if (p->val > root->val)
            return postorder(root->right, p, q);
        return root;
    }

    TreeNode *lowestCommonAncestor(TreeNode *root, TreeNode *p, TreeNode *q)
    {
        if (p->val > q->val)
        {
            swap(p, q);
        }
        return postorder(root, p, q);
    }
};
```

# 随想录

```cpp
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (root->val > p->val && root->val > q->val) {
            return lowestCommonAncestor(root->left, p, q);
        } else if (root->val < p->val && root->val < q->val) {
            return lowestCommonAncestor(root->right, p, q);
        } else return root;
    }
};
```
