# n皇后

Tags: 回溯
URL: https://leetcode.cn/problems/n-queens/description/

# 我的方法

这是一个排列问题，按层进行深度优先搜索，自然不会出现同行皇后的情况。

使用used记录在向下过程中哪些节点被遍历过了，自然不会出现同列皇后的情况。

只需要判断是否有对角线皇后的情况即可，对角线有两个方向，通过path数组记录当前已经出现的皇后的情况，然后从后往前遍历path数组，保证不出现对角线即可。

```cpp
class Solution
{
public:
    vector<string> element; // 用来输出答案的
    vector<string> path_element;
    vector<int> path;
    vector<vector<string>> result;
    vector<bool> used;

    bool is_valid(int depth, int col)
    {
        int j = col - 1;
        int k = col + 1;
        for (int i = depth - 1; i >= 0; i--)
        {
            if (path[i] == j||path[i] == k)
                return false;
            j--;
            k++;
        }
        return true;
    }

    void backtracking(int n, int depth)
    {
        if (depth == n)
        {
            result.push_back(path_element);
            return;
        }

        for (int i = 0; i < n; i++)
        {
            if (!used[i])
            {
                if (is_valid(depth, i))
                {
                    path.push_back(i);
                    path_element.push_back(element[i]);
                    used[i] = true;
                    backtracking(n, depth + 1);
                    used[i] = false;
                    path_element.pop_back();
                    path.pop_back();
                }
            }
        }
    }

    vector<vector<string>> solveNQueens(int n)
    {
        for (int i = 0; i < n; i++)
        {
            string tmp(n, '.');
            tmp[i] = 'Q';
            element.push_back(tmp);
        }
        used.resize(n, false);
        backtracking(n, 0);
        return result;
    }
};
```

# 随想录

```cpp
class Solution {
private:
vector<vector<string>> result;
// n 为输入的棋盘大小
// row 是当前递归到棋盘的第几行了
void backtracking(int n, int row, vector<string>& chessboard) {
    if (row == n) {
        result.push_back(chessboard);
        return;
    }
    for (int col = 0; col < n; col++) {
        if (isValid(row, col, chessboard, n)) { // 验证合法就可以放
            chessboard[row][col] = 'Q'; // 放置皇后
            backtracking(n, row + 1, chessboard);
            chessboard[row][col] = '.'; // 回溯，撤销皇后
        }
    }
}
bool isValid(int row, int col, vector<string>& chessboard, int n) {
    // 检查列
    for (int i = 0; i < row; i++) { // 这是一个剪枝
        if (chessboard[i][col] == 'Q') {
            return false;
        }
    }
    // 检查 45度角是否有皇后
    for (int i = row - 1, j = col - 1; i >=0 && j >= 0; i--, j--) {
        if (chessboard[i][j] == 'Q') {
            return false;
        }
    }
    // 检查 135度角是否有皇后
    for(int i = row - 1, j = col + 1; i >= 0 && j < n; i--, j++) {
        if (chessboard[i][j] == 'Q') {
            return false;
        }
    }
    return true;
}
public:
    vector<vector<string>> solveNQueens(int n) {
        result.clear();
        std::vector<std::string> chessboard(n, std::string(n, '.'));
        backtracking(n, 0, chessboard);
        return result;
    }
};
```
