# 904. 水果成篮

Tags: 双指针, 滑动窗口
URL: https://leetcode.cn/problems/fruit-into-baskets/description/

双指针，滑动窗口。

确定更新左指针的时机，右指针的时机，以及滑动窗口中有什么内容。

```cpp
class Solution
{
public:
    int totalFruit(std::vector<int> &fruits)
    {
        std::vector<int> bucket;
        int left = 0; // 左闭右开区间
        int right = 0;
        int result = 0;
        while (right < fruits.size())
        {
            if (bucket.size() < 2)
            {
                if (bucket.size() == 0 || bucket[0] != fruits[right])
                    bucket.push_back(fruits[right]);
                right++;
            }
            else if (bucket.size() == 2)
            {
                if (bucket[0] == fruits[right] || bucket[1] == fruits[right])
                    right++;
                else
                {
                    if (right - left > result)
                        result = right - left;
                    // 当前right指向的元素不在篮子里，需要更新left，删除篮子里面的一个元素。
                    int temp = fruits[right - 1]; // 记录当前元素的后一个元素,可以留在篮子里
                    if (temp == bucket[0])
                        bucket.erase(bucket.begin() + 1);
                    else
                        bucket.erase(bucket.begin());
                    // 更新left
                    int k = right - 1;
                    while (fruits[k] == temp)
                        k--;
                    left = k + 1;
                }
            }
        }
        if (right - left > result)
            result = right - left;
        return result;
    }
};
```

题解，使用了unordered map。

```cpp
class Solution
{
public:
    int totalFruit(vector<int> &fruits)
    {
        int n = fruits.size();
        unordered_map<int, int> cnt;
        int left = 0, ans = 0;
        for (int right = 0; right < n; ++right)
        {
            ++cnt[fruits[right]];
            while (cnt.size() > 2)
            {
                auto it = cnt.find(fruits[left]);
                --it->second;
                if (it->second == 0)
                {
                    cnt.erase(it);
                }
                ++left;
            }
            ans = max(ans, right - left + 1);
        }
        return ans;
    }
};
```
