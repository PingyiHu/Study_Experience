# 350. 两个数组的交集 II

Tags: 哈希表
URL: https://leetcode.cn/problems/intersection-of-two-arrays-ii/description/

# 力扣答案

```cpp
class Solution
{
public:
    vector<int> intersect(vector<int> &nums1, vector<int> &nums2)
    {
        if (nums1.size() > nums2.size())
        {
            return intersect(nums2, nums1);
        }
        unordered_map<int, int> m;
        for (int num : nums1)
        {
            ++m[num];
        }
        vector<int> intersection;
        for (int num : nums2)
        {
            if (m.count(num))
            {
                intersection.push_back(num);
                --m[num];
                if (m[num] == 0)
                {
                    m.erase(num);
                }
            }
        }
        return intersection;
    }
};
```

# 我的方法

使用map1记录num1中元素出现的次数。再用map2记录差值。

```cpp
class Solution
{
public:
    vector<int> intersect(vector<int> &nums1, vector<int> &nums2)
    {
        map<int, int> m1;
        vector<int> r;
        for (int i = 0; i < nums1.size(); i++)
        {
            if (m1.find(nums1[i]) == m1.end())
            {
                m1[nums1[i]] = 1;
            }
            else
            {
                m1[nums1[i]]++;
            }
        }
        map<int, int> m2(m1.begin(), m1.end());
        for (int i = 0; i < nums2.size(); i++)
        {
            if (m2.find(nums2[i]) != m2.end())
                m2[nums2[i]]--;
        }
        for (auto it1 = m1.begin(), it2 = m2.begin(); it1 != m1.end(); it1++, it2++)
        {
            int time = 0;
            if (it2->second < 0)
                time = it1->second;
            else
                time = it1->second - it2->second;
            for (int i = 0; i < time; i++)
                r.push_back(it1->first);
        }
        return r;
    }
};
```
