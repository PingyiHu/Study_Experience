# 20. 有效的括号

Tags: 栈
URL: https://leetcode.cn/problems/valid-parentheses/description/

```cpp
class Solution {
public:
    bool isValid(string s) {
        stack<int> st;
        for (int i = 0; i < s.size(); i++) {
            if (s[i] == '(' || s[i] == '{' || s[i] == '[')
                st.push(s[i]);
            else {
                if (st.empty())
                    return false;
                if ((st.top() == '(' && s[i] == ')') ||
                    (st.top() == '[' && s[i] == ']') ||
                    (st.top() == '{' && s[i] == '}')) {
                    st.pop();
                } else
                    return false;
            }
        }
        if (st.empty())
            return true;
        else
            return false;
    }
};
```
