# 332. 重新安排行程

Tags: 回溯
URL: https://leetcode.cn/problems/reconstruct-itinerary/description/

# 我的方法

排列机票之后，进行回溯查找。

```cpp
class Solution
{
public:
    vector<string> result;
    vector<string> tmp;
    vector<bool> used; // 记录是否该票已经用过了
    bool complete_flag = false;
    struct cmp
    {
        bool operator()(const vector<string> &a, const vector<string> &b) const
        {
            // 用 string 自己的 < 就行，等价于字典序
            return a[1] < b[1];
        }
    };
    void backtracking(vector<vector<string>> &tickets, string &pre)
    {
        if (complete_flag)
            return;
        if (tmp.size() == tickets.size() + 1)
        {
            complete_flag = true;
            result = tmp;
            return;
        }
        for (int i = 0; i < tickets.size(); i++)
        {
            if (!used[i] && tickets[i][0] == pre)
            {
                tmp.push_back(tickets[i][1]);
                used[i] = true;
                backtracking(tickets, tickets[i][1]);
                used[i] = false;
                tmp.pop_back();
            }
        }
    }

    vector<string> findItinerary(vector<vector<string>> &tickets)
    {
        sort(tickets.begin(), tickets.end(), cmp{});
        used.resize(tickets.size(), false);
        tmp.push_back("JFK");
        for (int i = 0; i < tickets.size(); i++)
        {
            if (tickets[i][0] == "JFK")
            {
                tmp.push_back(tickets[i][1]);
                used[i] = true;
                backtracking(tickets, tickets[i][1]);
                used[i] = false;
                tmp.pop_back();
            }
        }
        return result;
    }
};
```

# 随想录

**一个点对应的集合全部找全，然后DFS进行遍历。**

将每一个机场可以到达的机场进行如下的记录：

unordered_map<string, map<string, int>> targets

- 即unordered_map<出发机场, map<到达机场, 航班次数>> targets
- map自动将key按照字典序从小到大排列

```cpp
class Solution {
private:
// unordered_map<出发机场, map<到达机场, 航班次数>> targets
unordered_map<string, map<string, int>> targets;
bool backtracking(int ticketNum, vector<string>& result) {
    if (result.size() == ticketNum + 1) {
        return true;
    }
    for (pair<const string, int>& target : targets[result[result.size() - 1]]) {
        if (target.second > 0 ) { // 记录到达机场是否飞过了
            result.push_back(target.first);
            target.second--;
            if (backtracking(ticketNum, result)) return true;
            result.pop_back();
            target.second++;
        }
    }
    return false;
}
public:
    vector<string> findItinerary(vector<vector<string>>& tickets) {
        targets.clear();
        vector<string> result;
        for (const vector<string>& vec : tickets) {
            targets[vec[0]][vec[1]]++; // 记录映射关系
        }
        result.push_back("JFK"); // 起始机场
        backtracking(tickets.size(), result);
        return result;
    }
};
```
