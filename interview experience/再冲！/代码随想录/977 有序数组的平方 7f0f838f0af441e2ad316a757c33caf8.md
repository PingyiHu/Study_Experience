# 977. 有序数组的平方

Tags: 双指针
URL: https://leetcode.cn/problems/squares-of-a-sorted-array/submissions/526201083/

我的方法：头尾指针，当需要找的东西在两边，不在数组中间时，就这么干。

```cpp
class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        int i = 0;
        int j = nums.size() - 1;
        int k = nums.size() - 1;
        vector<int> temp = nums;
        while (i <= j) {
            if (pow(nums[i], 2) > pow(nums[j], 2)) {
                temp[k--] = pow(nums[i++], 2);
            } else
                temp[k--] = pow(nums[j--], 2);
        }
        return temp;
    }
};
```

答案双指针法：找正负交界，分为两个数组做归并排序。

```cpp
class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {
        int n = nums.size();
        int negative = -1;
        for (int i = 0; i < n; ++i) {
            if (nums[i] < 0) {
                negative = i;
            } else {
                break;
            }
        }

        vector<int> ans;
        int i = negative, j = negative + 1;
        while (i >= 0 || j < n) {
            if (i < 0) {
                ans.push_back(nums[j] * nums[j]);
                ++j;
            }
            else if (j == n) {
                ans.push_back(nums[i] * nums[i]);
                --i;
            }
            else if (nums[i] * nums[i] < nums[j] * nums[j]) {
                ans.push_back(nums[i] * nums[i]);
                --i;
            }
            else {
                ans.push_back(nums[j] * nums[j]);
                ++j;
            }
        }

        return ans;
    }
};
```
