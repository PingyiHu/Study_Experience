# 202. 快乐数

Tags: 哈希表, 基础库使用
URL: https://leetcode.cn/problems/happy-number/description/

我的答案

```cpp
class Solution
{
public:
    bool isHappy(int n)
    {
        int multiply[10] = {0};
        unordered_set<int> appear;
        for (int i = 0; i < 10; i++)
        {
            multiply[i] = i * i;
        }
        while (1)
        {
            int temp = 0;
            while (n)
            {
                temp += multiply[n % 10];
                n /= 10;
            }
            if (temp == 1)
                return true;
            else if (appear.find(temp) != appear.end())
                return false;
            else
            {
                appear.insert(temp);
                n = temp;
            }
        }
    }
};
```
