# 150. 逆波兰表达式求值

Tags: 栈
URL: https://leetcode.cn/problems/evaluate-reverse-polish-notation/

# 我的方法

将数字保存到栈内，遇到运算符号，就运算后把结果放到栈内。

```cpp
class Solution {
public:
    int string2number(string token) {
        int res = 0;
        int i = 0;
        if (token[0] == '-')
            i += 1;
        for (; i < token.size(); i++) {
            res = res * 10 + token[i]-'0';
        }
        if (token[0] == '-')
            return -1 * res;
        else
            return res;
    }

    int evalRPN(vector<string>& tokens) {
        stack<int> st2;
        for (int i = 0; i < tokens.size(); i++) {
            if (tokens[i].size() == 1 &&
                (tokens[i][0] < '0' || tokens[i][0] > '9')) {
                int num1 = st2.top();
                st2.pop();
                int num2 = st2.top();
                st2.pop();
                if (tokens[i][0] == '+') {
                    st2.push(num2 + num1);
                } else if (tokens[i][0] == '-') {
                    st2.push(num2 - num1);
                } else if (tokens[i][0] == '*') {
                    st2.push(num2 * num1);
                } else {
                    st2.push(num2 / num1);
                }
            } else {
                st2.push(string2number(tokens[i]));
            }
        }
        return st2.top();
    }
};
```
