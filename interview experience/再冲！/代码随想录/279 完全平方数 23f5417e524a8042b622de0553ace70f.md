# 279.完全平方数

Tags: 动态规划
URL: https://leetcode.cn/problems/perfect-squares/submissions/648102634/

# 我的方法

比n大的完全平方数是无法组成n的，因此先对n开方取整。

```cpp
class Solution {
public:
    int numSquares(int n) {
        int max_num = int(sqrt(n));
        vector<int> nums;
        for (int i = 1; i <= max_num; i++)
            nums.push_back(i * i);
        vector<int> dp(n + 1, INT_MAX);
        dp[0] = 0;
        for (int i = 0; i < nums.size(); i++) {
            for (int j = 1; j <= n; j++) {
                if (j - nums[i] >= 0)
                    dp[j] = min(dp[j], dp[j - nums[i]] + 1);
            }
        }
        return dp[n];
    }
};
```

# 随想录

```cpp
// 版本一
class Solution {
public:
    int numSquares(int n) {
        vector<int> dp(n + 1, INT_MAX);
        dp[0] = 0;
        for (int i = 0; i <= n; i++) { // 遍历背包
            for (int j = 1; j * j <= i; j++) { // 遍历物品
                dp[i] = min(dp[i - j * j] + 1, dp[i]);
            }
        }
        return dp[n];
    }
};
```
