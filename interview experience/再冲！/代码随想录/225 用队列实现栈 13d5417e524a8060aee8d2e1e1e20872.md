# 我的方法

只要一个队列不为空，我让其不断出栈到只剩下一个数据，这个队尾元素直接返回就行，两个队列不停相互倒。

```cpp
class MyStack {
public:
    MyStack() {}

    void push(int x) {
        if (q1.empty() && q2.empty())
            q1.push(x);
        else if (!q1.empty())
            q1.push(x);
        else
            q2.push(x);
    }

    int pop() {
        int temp;
        if (!q1.empty()) {
            while (q1.size() != 1) {
                q2.push(q1.front());
                q1.pop();
            }
            temp = q1.front();
            q1.pop();
        } else {
            while (q2.size() != 1) {
                q1.push(q2.front());
                q2.pop();
            }
            temp = q2.front();
            q2.pop();
        }
        return temp;
    }

    int top() {
        int temp = this->pop();
        this->push(temp);
        return temp;
    }

    bool empty() { return q1.empty() && q2.empty(); }

private:
    queue<int> q1;
    queue<int> q2;
};

/**
 * Your MyStack object will be instantiated and called as such:
 * MyStack* obj = new MyStack();
 * obj->push(x);
 * int param_2 = obj->pop();
 * int param_3 = obj->top();
 * bool param_4 = obj->empty();
 */
```
